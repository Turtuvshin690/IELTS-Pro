# IELTS Pro Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship downloadable Electron PC app covering IELTS Reading/Listening/Writing/Speaking with timed mocks, local SQLite + filesystem storage, and BYOK NVIDIA NIM (Nemotron 3.5 Lightning for Writing/Speaking scoring + Parakeet/Magpie for Speaking).

**Architecture:** Electron Main (Node) proxies OpenAI-compatible NIM calls with BYOK key from `safeStorage`; Renderer is React 18 + Vite + Tailwind + Zustand + React Router (hash). All data local in `%APPDATA%/IELTS Pro` via `better-sqlite3` WAL and `audio/` cache; no backend. `electron-builder` produces NSIS/MSI/portable.

**Tech Stack:** Electron 30 + electron-vite, Vite 5, React 18, TypeScript 5 strict, Tailwind 3, Zustand, React Router v6, better-sqlite3, Vitest, Playwright (electron), MSW, electron-updater, WaveSurfer-like audio, MediaRecorder.

## Global Constraints

- Everything local except model inference — no backend, no cloud DB, no user accounts.
- BYOK only — key via `build.nvidia.com`, stored encrypted via `safeStorage`, never in renderer/plaintext.
- Nemotron 3.5 Lightning is text-only (30B total / 3B active, 256K ctx, `temperature: 1.0, top_p: 0.95`, reasoning parser `nemotron_v3`) — used only for text scoring; Speaking P band is heuristic.
- Parakeet 1.1b CTC en-US ASR (`parakeet-1-1b-ctc-en-us:1.5.3`, streaming+offline, HTTP :9000) + Magpie TTS Multilingual (`magpie-tts-multilingual:1.10.0`) for Speaking.
- Windows 10/11 support via electron-builder NSIS/MSI/portable.
- Unlimited retakes — no artificial limits, paginated local history.
- Node 20, Electron 30, TypeScript 5 strict, ESLint + Prettier.
- CSP `default-src 'self'` in renderer; IPC allowlist only.

```ts
// src/main/ipc/vault.ts
import { ipcMain, safeStorage, app } from 'electron';
import fs from 'fs'; import path from 'path';
const KEY_FILE = () => path.join(app.getPath('userData'), 'vault.bin');
export function encryptKey(k: string) { return safeStorage.encryptString(k); }
export function decryptKey(b: Buffer) { return safeStorage.decryptString(b); }
export async function validateKey(key: string) {
  const r = await fetch('https://integrate.api.nvidia.com/v1/models', { headers: { Authorization: `Bearer ${key}` } });
  return r.ok;
}
export function registerVaultIpc() {
  ipcMain.handle('vault:set', (_e, key:string) => {
    const enc = encryptKey(key);
    fs.writeFileSync(KEY_FILE(), enc);
    return true;
  });
  ipcMain.handle('vault:get', () => {
    if (!fs.existsSync(KEY_FILE())) return null;
    return decryptKey(fs.readFileSync(KEY_FILE()));
  });
  ipcMain.handle('vault:validate', (_e, key:string) => validateKey(key));
}
```

```ts
// src/lib/nim/nemotron.ts
import { WRITING_SYSTEM } from './prompts';
export const NIM_BASE = 'https://integrate.api.nvidia.com';
export function buildChatPayload(messages:any[]) {
  return { model: 'nvidia/nemotron-3.5-lightning-30b-a3b', messages, temperature: 1.0, top_p: 0.95, max_tokens: 2048 };
}
export async function chat(key:string, messages:any[]) {
  const r = await fetch(`${NIM_BASE}/v1/chat/completions`, {
    method:'POST', headers:{ Authorization:`Bearer ${key}`, 'Content-Type':'application/json' },
    body: JSON.stringify(buildChatPayload(messages))
  });
  if (!r.ok) throw new Error(`NIM ${r.status}: ${await r.text()}`);
  const j = await r.json();
  return j.choices[0].message.content as string;
}

// src/lib/nim/parakeet.ts — Speech NIM HTTP :9000 (cloud)
export async function transcribe(key:string, wav:Buffer) {
  const form = new FormData(); form.append('file', new Blob([wav]), 'audio.wav'); form.append('model','parakeet-1-1b-ctc-en-us');
  const r = await fetch('https://integrate.api.nvidia.com/v1/audio/transcriptions', { method:'POST', headers:{ Authorization:`Bearer ${key}` }, body: form as any });
  if (!r.ok) throw new Error(`ASR ${r.status}`);
  const j = await r.json(); return j.text as string;
}

// src/lib/nim/magpie.ts
export async function synthesize(key:string, text:string, voice='Magpie-Multilingual.en-US.Ava') {
  const r = await fetch('https://integrate.api.nvidia.com/v1/audio/speech', { method:'POST', headers:{ Authorization:`Bearer ${key}`, 'Content-Type':'application/json' }, body: JSON.stringify({ model: voice, input: text }) });
  if (!r.ok) throw new Error(`TTS ${r.status}`);
  return Buffer.from(await r.arrayBuffer());
}
```

```ts
// src/lib/band/calculator.ts
const READING_MAP: Record<number, number> = { 39:9,37:8.5,35:8,33:7.5,30:7,27:6.5,23:6,19:5.5,15:5,13:4.5,10:4,8:3.5,6:3,4:2.5 };
export function rawToBand(raw:number, total:number){
  if(total!==40) return Math.round((raw/total)*9*2)/2;
  const keys = Object.keys(READING_MAP).map(Number).sort((a,b)=>b-a);
  for(const k of keys) if(raw>=k) return READING_MAP[k];
  return 0;
}
export function overallBand(scores:number[]){ const avg = scores.reduce((a,b)=>a+b,0)/scores.length; return Math.round(avg*2)/2; }

import { useEffect, useRef, useState } from 'react';
export function useTimer(durationSec:number, onExpire:()=>void){
  const [remaining, setRemaining] = useState(durationSec);
  const ref = useRef<number | null>(null);
  useEffect(()=>{ ref.current = window.setInterval(()=> setRemaining(r=>{ if(r<=1){ clearInterval(ref.current!); onExpire(); return 0;} return r-1;}),1000); return ()=> clearInterval(ref.current!);},[]);
  return { remaining, formatted: `${String(Math.floor(remaining/60)).padStart(2,'0')}:${String(remaining%60).padStart(2,'0')}` };
}
```

```ts
// src/lib/import/validator.ts
export type ImportPayload = { tests:any[] };
const Q_TYPES = ['MCQ','TFNG','YNNG','MATCH_HEADING','MATCH_FEATURE','SENTENCE_COMPLETION','SUMMARY_COMPLETION','DIAGRAM_LABEL','SHORT_ANSWER','MAP_LABEL','FORM_COMPLETION'];
export function validateImport(p:any){
  const errors:string[]=[]; if(!p.tests?.length) errors.push('No tests');
  for(const t of p.tests||[]){ if(!t.id) errors.push('Test missing id'); for(const s of t.sections||[]){ for(const q of s.questions||[]){ if(!Q_TYPES.includes(q.qType)) errors.push(`Bad qType ${q.qType}`); if(q.answer==null) errors.push(`Q ${q.id} missing answer`);} } }
  return { ok: errors.length===0, errors };
}
```
